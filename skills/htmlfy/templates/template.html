rate_limiter.py 
def allow(self, key):
    bucket = self.buckets[key]
    bucket.refill()              # add tokens for elapsed time
    if bucket.tokens >= 1:
        bucket.tokens -= 1       # spend one token
        return True
    return False                 # rate limited
